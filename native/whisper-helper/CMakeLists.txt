cmake_minimum_required(VERSION 3.24)
project(computercat_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
include(FetchContent)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/dependencies.lock.json" LOCK)
foreach(DEP whisper json)
  string(JSON URL GET "${LOCK}" ${DEP} url)
  string(JSON HASH GET "${LOCK}" ${DEP} sha256)
  FetchContent_Declare(${DEP} URL "${URL}" URL_HASH "SHA256=${HASH}" DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
endforeach()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_IS_DEV OFF CACHE BOOL "" FORCE)
option(COMPUTERCAT_AVX2 "Build the guarded fast CPU variant" OFF)
foreach(FLAG NATIVE OPENMP AVX AVX2 FMA F16C SSE42 BMI2 CUDA)
  set(GGML_${FLAG} OFF CACHE BOOL "" FORCE)
endforeach()
if(COMPUTERCAT_AVX2)
  foreach(FLAG AVX AVX2 FMA F16C BMI2)
    set(GGML_${FLAG} ON CACHE BOOL "" FORCE)
  endforeach()
endif()
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(whisper json)
add_executable(computercat-whisper src/main.cpp)
target_link_libraries(computercat-whisper PRIVATE whisper nlohmann_json::nlohmann_json)
target_compile_definitions(computercat-whisper PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(computercat-whisper PRIVATE /utf-8 /W4)
endif()
if(COMPUTERCAT_AVX2)
  target_compile_definitions(computercat-whisper PRIVATE COMPUTERCAT_AVX2)
  install(TARGETS computercat-whisper RUNTIME DESTINATION cpu-avx2 COMPONENT helper)
else()
  install(TARGETS computercat-whisper RUNTIME DESTINATION cpu COMPONENT helper)
endif()


option(COMPUTERCAT_TESTS "Build the offline injected-inference test executable" ON)
if(COMPUTERCAT_TESTS AND NOT COMPUTERCAT_AVX2)
  add_executable(computercat-whisper-test src/main.cpp)
  target_link_libraries(computercat-whisper-test PRIVATE whisper nlohmann_json::nlohmann_json)
  target_compile_definitions(computercat-whisper-test PRIVATE COMPUTERCAT_TEST_ENGINE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
